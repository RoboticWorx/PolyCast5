Frames gathered from https://github.com/Rezmason/matrix if you're interested.

I ended up cutting out all the odd frames to save some memory.

        <style>
            @supports (padding-top: env(safe-area-inset-top)) {
                body {
                    padding: 0;
                    height: calc(100% + env(safe-area-inset-top));
                }
            }
            body {
                background: black;
                overflow: hidden;
                margin: 0;

                font-family: monospace;

                text-align: center;
            }

                height: 100vh;
            }

            }

            .notice {
                margin-top: 10em;
                animation: fadeInAnimation ease 3s;
                animation-iteration-count: 1;
                animation-fill-mode: forwards;
            }

            @keyframes fadeInAnimation {
                0% {
                    opacity: 0;
                }
                100% {
                    opacity: 1;
                }
            }

            .pill {

    <body>
        <script>
        (function(){
          const orig = HTMLCanvasElement.prototype.getContext;
          HTMLCanvasElement.prototype.getContext = function(type, attrs) {
            if ((type === 'webgl' || type === 'webgl2')) {
              attrs = Object.assign({}, attrs, { preserveDrawingBuffer: true });
            }
            return orig.call(this, type, attrs);
          };
        })();
      </script>
    
      <!-- 1) Force URL params -->
      <script>
        (function() {
          const q = new URLSearchParams(location.search);
          q.set('version',      'classic');
          q.set('skipIntro',    'true');
          q.set('numColumns',   '40');
          q.set('bloomStrength','0.4');
          q.set('loops',        'true');
          q.set('fps',          '20');
          if (location.search !== '?' + q.toString()) {
            location.search = q.toString();
          }
        })();
      </script>
    
      <!-- 2) FileSaver.js & JSZip -->
      <script src="https://cdn.jsdelivr.net/npm/file-saver@2.0.5/dist/FileSaver.min.js"></script>
      <script src="https://cdnjs.cloudflare.com/ajax/libs/jszip/3.7.1/jszip.min.js"></script>
    
      <!-- 3) Matrix demo -->
      <script type="module" src="js/main.js"></script>
    
      <!-- 4) Capture logic -->
      <script>
        window.addEventListener('load', () => {
    
          // Wait for the real (non-zero) canvas
          function findCanvas() {
            return Array.from(document.querySelectorAll('canvas'))
                        .find(c => c.width > 0 && c.height > 0);
          }
          function waitForCanvas(cb) {
            const c = findCanvas();
            if (c) cb(c);
            else   setTimeout(() => waitForCanvas(cb), 50);
          }
    
          waitForCanvas(mainCanvas => {
            // offscreen canvas for 240×135
            const w = 240, h = 135;
            const rs = document.createElement('canvas');
            rs.width = w; rs.height = h;
            const ctx = rs.getContext('2d');
    
            const zip       = new JSZip();
            const FPS       = 20;             // match ?fps=20
            const CYCLE_SEC = 6;
            const WANT      = 100;             // change this for more frames
            const PERIOD_MS = (CYCLE_SEC * 1000) / WANT;
    
            let kept = 0, intervalId;
    
            function snap() {
              ctx.drawImage(
                mainCanvas,
                0, 0,
                mainCanvas.width, mainCanvas.height,
                0, 0,
                w, h
              );
              rs.toBlob(blob => {
                kept++;
                zip.file(
                  `anim_matrix_rain_${kept}.png`,
                  blob,
                  { compression: 'STORE' }
                );
                if (kept === WANT) {
                  clearInterval(intervalId);
                  zip.generateAsync({ type:'blob', compression:'STORE' })
                     .then(zb => saveAs(zb, 'matrix_frames_240x135.zip'));
                }
              }, 'image/png');
            }
    
            // warm-up then align to loop boundary
            setTimeout(() => {
              const now   = performance.now();
              const delay = (CYCLE_SEC * 1000) - (now % (CYCLE_SEC * 1000));
              setTimeout(() => {
                snap();                              // first frame
                intervalId = setInterval(snap, PERIOD_MS);
              }, delay + 100);
            }, 500);
          });
    
        });
    
    // http://localhost:8000/index.html
    </script>
    </body>
</html>

        const now   = performance.now();
